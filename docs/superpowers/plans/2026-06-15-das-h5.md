    n = 1000
    trend = (3.0 * np.arange(n) + 5.0)[:, None]
    out = detrend(trend.copy().astype(float), type="linear")
    assert np.allclose(out, 0.0, atol=1e-6)


def test_bandpass_suppresses_out_of_band():
    delta = 0.01  # 100 Hz sampling
    n = 4096
    t = np.arange(n) * delta
    inband = np.sin(2 * np.pi * 2.0 * t)    # 2 Hz, kept
    outband = np.sin(2 * np.pi * 40.0 * t)  # 40 Hz, rejected
    data = (inband + outband)[:, None]
    out = filter(data.copy(), delta=delta, type="bandpass",
                 freq=(1.0, 10.0), order=4, zerophase=True)
    # residual after removing the in-band part is far smaller than the out-of-band input
    assert np.std(out[:, 0] - inband) < 0.2 * np.std(outband)


def test_decimate_changes_rate_and_length():
    delta = 0.01
    n = 1000
    data = np.random.randn(n, 3)
    out = decimate(data, delta=delta, factor=5)
    assert out.shape == (n // 5, 3)

import numpy as np
import h5py
from obspy import UTCDateTime

from das_h5.io.h5_reader import make_dbfile, read_h5
from das_h5.io.writer import save_npz


def _make_fake_h5(path, start_utc, npts=1000, nch=5, rate=1000.0, gauge=10.0):
    t0_us = int(start_utc.timestamp * 1e6)
    with h5py.File(path, "w") as h5:
        raw = h5.create_group("Acquisition/Raw[0]")
        data = (np.arange(npts)[:, None] * np.ones((1, nch))).astype(np.float32)
        raw.create_dataset("RawData", data=data)
        raw.create_dataset("RawDataTime", data=np.array([t0_us], dtype=np.int64))
        raw.attrs["OutputDataRate"] = rate
        h5["Acquisition"].attrs["GaugeLength"] = gauge


def test_make_dbfile_and_read_h5_roundtrip(tmp_path):
    start = UTCDateTime("2024-01-01T00:00:00")
    _make_fake_h5(str(tmp_path / "das_000.h5"), start, npts=1000, nch=5, rate=1000.0)

    db = make_dbfile(str(tmp_path))
    assert len(db) == 1

    data, dt = read_h5(db_list=db, time_flag=start, begin_time=0, end_time=0.5,
                       channel_begin=0, channel_end=4)
    assert data.shape[1] == 5
    assert abs(dt - 0.001) < 1e-9
    assert 490 <= data.shape[0] <= 510


def test_save_npz_roundtrip(tmp_path):
    data = np.random.randn(200, 4).astype(np.float32)
    out = str(tmp_path / "trace.npy")
    save_npz(data, outnpy=out)
    reloaded = np.load(out)
    assert reloaded.shape == data.shape
    assert np.allclose(reloaded, data)

import numpy as np
from obspy import UTCDateTime

from das_h5.psd.compute import das_psd


def test_das_psd_peaks_at_input_frequency():
    delta = 0.01           # 100 Hz sampling
    n = 2000               # 20 s
    f0 = 10.0
    t = np.arange(n) * delta
    data = np.sin(2 * np.pi * f0 * t)[:, None].astype(np.float64)

    fss, psds, times, maxv = das_psd(
        das_data=data, delta=delta, wind_len=2.0, wind_step=2.0,
        minfreq=[1.0], maxfreq=[40.0], ch1=0,
        starttime=UTCDateTime(0), scaling="density")

    freqs = fss[0][0]            # band 0 frequency axis of first segment
    power = psds[0][0][:, 0]     # band 0 PSD of channel 0
    peak_freq = freqs[int(np.argmax(power))]
    assert abs(peak_freq - f0) < 1.0
    assert len(times) >= 1
